/*
====================================
DATA CLASSES - ÇOK ÖNEMLİ!
====================================

Data class'lar veri tutmak için özel olarak tasarlanmış class'lardır.
"data" keyword'ü eklediğinizde Kotlin otomatik olarak şunları üretir:
- equals() ve hashCode()
- toString() 
- copy()
- componentN() fonksiyonları

Android'de Model class'ları her zaman data class olur!
API'den gelen JSON'ları data class'lara map edersiniz.
*/

data class Urun(
    val id: Int,
    val ad: String,
    var fiyat: Double,
    val stok: Int
)

// Data class'ın sağladığı özellikler:
fun dataClassOzellikleri() {
    val urun1 = Urun(1, "Laptop", 15000.0, 10)
    val urun2 = Urun(1, "Laptop", 15000.0, 10)
    val urun3 = Urun(2, "Mouse", 150.0, 50)
    
    // 1. toString() otomatik
    println(urun1)  
    // Çıktı: Urun(id=1, ad=Laptop, fiyat=15000.0, stok=10)
    // Normal class'ta: Urun@2f4d3709 gibi anlamsız çıktı verirdi
    
    // 2. equals() otomatik - içeriği karşılaştırır
    println(urun1 == urun2)  // true - içerikleri aynı
    println(urun1 == urun3)  // false - içerikleri farklı
    
    // 3. copy() fonksiyonu - mevcut nesneden kopyalayıp değişiklik yapma
    val indirimliFiyat = urun1.copy(fiyat = 12000.0)
    // Sadece fiyat değişti, diğer değerler aynı kaldı
    println(indirimliFiyat)
    // Çıktı: Urun(id=1, ad=Laptop, fiyat=12000.0, stok=10)
    
    // 4. Destructuring - komponenlere ayırma
    val (id, ad, fiyat, stok) = urun1
    println("Ürün adı: $ad, Fiyatı: $fiyat")
}

/*
====================================
SEALED CLASSES - STATE YÖNETİMİ İÇİN KRİTİK!
====================================

Sealed class'lar sınırlı sayıda alt tipe sahip olabilen class'lardır.
Enum'a benzer ama her alt tip farklı property'lere sahip olabilir.

Compose ve modern Android'de State yönetimi için MUTLAKA kullanılır!
When kullanıldığında tüm durumları kontrol etmeye zorlar - bu harika!
*/

// UI State için sealed class - Her ekranın durumları
sealed class UiState {
    object Loading : UiState()  // Yükleniyor - veri yok
    data class Success(val data: List<String>) : UiState()  // Başarılı - veri var
    data class Error(val message: String) : UiState()  // Hata - mesaj var
    object Empty : UiState()  // Boş state - liste boş geldi
}

// Kullanım - State'e göre UI gösterme
fun ekraniGoster(state: UiState) {
    when (state) {
        is UiState.Loading -> {
            println("Yükleniyor animasyonu göster")
        }
        is UiState.Success -> {
            println("Veriyi göster: ${state.data}")
        }
        is UiState.Error -> {
            println("Hata mesajı göster: ${state.message}")
        }
        is UiState.Empty -> {
            println("Boş liste mesajı göster")
        }
        // else gerekmez! Tüm durumlar kapsandı
    }
}

/*
====================================
COMPANION OBJECT - STATIC MUADILI
====================================

Kotlin'de static keyword'ü yok! Onun yerine companion object var.
Class'ın tüm instance'larında ortak olan değerler/fonksiyonlar için kullanılır.

Java'da: ClassName.staticMethod()
Kotlin'de: ClassName.companionMethod()
*/

class VeriTabani {
    companion object {
        // Sabitler - tüm instance'larda aynı
        const val DB_NAME = "kullanici_db"
        const val DB_VERSION = 1
        
        // Singleton instance
        private var INSTANCE: VeriTabani? = null
        
        // Factory method - nesne oluşturucu
        fun getInstance(): VeriTabani {
            if (INSTANCE == null) {
                INSTANCE = VeriTabani()
            }
            return INSTANCE!!
        }
    }
    
    fun kaydet(veri: String) {
        println("$veri veritabanına kaydedildi")
    }
}

// Kullanım:
fun companionKullanimi() {
    // Static gibi doğrudan erişim
    println(VeriTabani.DB_NAME)
    
    // Singleton pattern
    val db = VeriTabani.getInstance()
    db.kaydet("Test verisi")
}

/*
COMPANION OBJECT - FACTORY PATTERN
Nesne oluşturmak için alternatif yollar sağlar
*/

class Kullanici private constructor(
    val id: Int,
    val ad: String,
    val email: String
) {
    companion object {
        // Factory metodlar - farklı yollarla nesne oluşturma
        fun emailIleOlustur(email: String): Kullanici {
            // Email'den kullanıcı adını çıkar
            val kullaniciAdi = email.substringBefore('@')
            return Kullanici(
                id = 0,
                ad = kullaniciAdi,
                email = email
            )
        }
        
        fun misafirOlustur(): Kullanici {
            return Kullanici(
                id = -1,
                ad = "Misafir",
                email = "misafir@temp.com"
            )
        }
    }
}

/*
====================================
OBJECT - SINGLETON
====================================

"object" keyword'ü ile singleton oluşturursunuz.
Uygulama boyunca sadece 1 instance olur.

Ne zaman kullanılır?
- Database bağlantıları
- Network client'lar (Retrofit instance)
- Ayar yöneticileri
- Logger
*/

object AyarYoneticisi {
    private val ayarlar = mutableMapOf<String, Any>()
    
    fun ayarKoy(anahtar: String, deger: Any) {
        ayarlar[anahtar] = deger
    }
    
    fun ayarAl(anahtar: String): Any? {
        return ayarlar[anahtar]
    }
    
    fun tumAyarlariTemizle() {
        ayarlar.clear()
    }
}

// Kullanım - new keyword'ü YOK!
fun objectKullanimi() {
    AyarYoneticisi.ayarKoy("tema", "karanlik")
    AyarYoneticisi.ayarKoy("dil", "TR")
    
    val tema = AyarYoneticisi.ayarAl("tema")
    println("Aktif tema: $tema")
}

/*
OBJECT EXPRESSIONS - ANONİM SINIFLAR
Interface implementation için kullanılır (Java'daki anonymous class gibi)
*/

interface ClickListener {
    fun onClick(view: String)
}

fun anonymousOrnek() {
    // Java'da new ClickListener() { ... } şeklinde yazılır
    // Kotlin'de object : ClassName şeklinde
    
    val listener = object : ClickListener {
        override fun onClick(view: String) {
            println("$view tıklandı")
        }
    }
    
    listener.onClick("Buton")
}

/*
====================================
ÖZET VE GERÇEK KULLANIM
====================================

1. DATA CLASS: 
   - API modelleri (User, Product, Response)
   - RecyclerView/LazyColumn elemanları
   - Intent/Bundle ile veri geçişi

2. SEALED CLASS:
   - UI State (Loading, Success, Error)
   - Navigation Events
   - User Actions/Events

3. COMPANION OBJECT:
   - Sabitler (API_KEY, BASE_URL)
   - Factory metodlar
   - Static utility fonksiyonlar

4. OBJECT:
   - Singleton (Database, NetworkClient)
   - Anonymous interface implementation

Android projelerinde bunların hepsini kullanacaksınız!
Özellikle Compose'da sealed class + data class kombinasyonu standarttır.
*/
