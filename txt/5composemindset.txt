/*
====================================
COMPOSE NEDİR VE NEDEN DEVRİM NİTELİĞİNDE?
====================================

Jetpack Compose, Android'in modern UI toolkit'idir. 2021'de stable oldu.
XML layout'ların yerine Kotlin kodu ile UI yazarsınız.

Eski sistem (XML):
- activity_main.xml'de UI tanımla
- MainActivity.kt'de findViewById ile bağla
- Değişiklik için view'ları manuel güncelle
- 2 farklı dosya, 2 farklı dil (XML + Kotlin)

Compose ile:
- Her şey Kotlin
- Tek dosyada UI ve logic
- Otomatik güncelleme (Reactive)
- Çok daha az kod
*/

/*
====================================
DECLARATIVE vs IMPERATIVE UI
====================================

Bu iki yaklaşım arasındaki fark Compose'u anlamak için KRİTİK!

IMPERATIVE (Zorunlu/Emirli) - XML/View System:
"Nasıl" yapılacağını adım adım söylersiniz.
"TextView'ı bul, textini değiştir, rengini kırmızı yap, görünür yap"

DECLARATIVE (Bildirimsel) - Compose:
"Ne" istediğinizi söylersiniz, nasıl yapılacağı Compose'un işi.
"Eğer hata varsa kırmızı text göster, yoksa yeşil text göster"
*/

// IMPERATIVE ÖRNEK - Eski Yöntem (View System)
class MainActivity : Activity() {
    private lateinit var textView: TextView
    private lateinit var button: Button
    private var counter = 0
    
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)
        
        textView = findViewById(R.id.textView)
        button = findViewById(R.id.button)
        
        // Butona tıklandığında - ADIM ADIM TALİMATLAR
        button.setOnClickListener {
            counter++                              // 1. Counter'ı arttır
            textView.text = "Sayı: $counter"      // 2. Text'i güncelle
            if (counter > 5) {                    // 3. Koşulu kontrol et
                textView.setTextColor(Color.RED)  // 4. Rengi değiştir
            } else {
                textView.setTextColor(Color.GREEN) // 5. Veya bu rengi kullan
            }
        }
    }
}

// DECLARATIVE ÖRNEK - Compose Yöntemi
@Composable
fun CounterScreen() {
    // State - değişebilir veri
    var counter by remember { mutableStateOf(0) }
    
    // UI - state'e göre otomatik güncellenir
    Column {
        Text(
            text = "Sayı: $counter",
            color = if (counter > 5) Color.Red else Color.Green  // Sadece "ne" istediğimizi söylüyoruz
        )
        
        Button(onClick = { counter++ }) {  // Counter değişince UI otomatik güncellenir!
            Text("Arttır")
        }
    }
}

/*
Farkı görüyor musunuz?
- Imperative: "Text'i bul, değiştir, rengini ayarla" - NASIL
- Declarative: "Counter > 5 ise kırmızı olsun" - NE

Compose'da state değişince UI otomatik yeniden çizilir (recomposition).
Manuel güncelleme yok!
*/

/*
====================================
COMPOSABLE FUNCTIONS
====================================

Composable function = UI parçası üreten fonksiyon
@Composable annotation'ı ile işaretlenir.

Kurallar:
1. @Composable annotation'ı olmalı
2. İsim büyük harfle başlamalı (convention)
3. Nothing döndürür (Unit)
4. Side effect'lerden kaçınılmalı
5. Idempotent olmalı (aynı input = aynı UI)
*/

// Basit Composable
@Composable
fun Greeting(name: String) {
    Text(text = "Merhaba $name!")
}

// Composable'lar diğer Composable'ları çağırabilir
@Composable
fun WelcomeScreen() {
    Column {
        Greeting("Ali")           // Composable çağırma
        Greeting("Veli")         
        Button(onClick = {}) {   // Button da bir Composable
            Text("Tıkla")        // Text de bir Composable
        }
    }
}

/*
COMPOSABLE'LAR NORMAL FONKSİYON DEĞİLDİR!

Normal fonksiyon:
fun topla(a: Int, b: Int) = a + b
val sonuc = topla(3, 5)  // Hemen çalışır, 8 döner

Composable fonksiyon:
@Composable
fun UserCard(user: User) { ... }
// UserCard(user)  // HATA! Composable sadece Composable içinde çağrılabilir

Composable'lar:
- Compose compiler tarafından özel işlenir
- UI tree'ye node ekler
- Recomposition'a katılır
- Sadece Composable scope'ta çağrılabilir
*/

/*
====================================
RECOMPOSITION - COMPOSE'UN KALBİ
====================================

Recomposition = State değişince UI'ın ilgili kısımlarının yeniden çizilmesi

Compose akıllıdır! Tüm ekranı yeniden çizmez, sadece değişen kısımları çizer.
Bu nasıl çalışır?
*/

@Composable
fun UserProfile() {
    // State tanımlama - değişebilir veri
    var userName by remember { mutableStateOf("Ali") }
    var userAge by remember { mutableStateOf(25) }
    var showDetails by remember { mutableStateOf(false) }
    
    Column {
        // Bu Text sadece userName değişince recompose olur
        Text("İsim: $userName")
        
        // Bu Text sadece userAge değişince recompose olur  
        Text("Yaş: $userAge")
        
        Button(onClick = { showDetails = !showDetails }) {
            Text(if (showDetails) "Gizle" else "Göster")
        }
        
        // Bu blok sadece showDetails değişince değerlendirilir
        if (showDetails) {
            UserDetailsSection()  // Koşullu composition
        }
    }
}

/*
Recomposition Süreci:
1. State değişir (userName = "Veli")
2. Compose değişikliği algılar
3. Sadece userName kullanan Composable'ları bulur
4. Onları yeniden çalıştırır (recompose)
5. UI güncellenir

ÖNEMLİ: Recomposition çok sık olabilir!
- Animasyonlarda her frame'de (60 fps = saniyede 60 kez)
- Scroll'da sürekli
- Text input'ta her karakter girişinde

Bu yüzden Composable'lar:
- Hızlı olmalı
- Side effect içermemeli
- Ağır işlem yapmamalı
*/

/*
====================================
REMEMBER - STATE'İ HATIRLAMAK
====================================

Recomposition'da Composable yeniden çalışır.
Normal değişkenler sıfırlanır! remember ile korunur.
*/

@Composable
fun WrongCounter() {
    var count = 0  // YANLIŞ! Her recomposition'da 0 olur
    
    Button(onClick = { count++ }) {
        Text("Sayı: $count")  // Her zaman 0 gösterir!
    }
}

@Composable
fun CorrectCounter() {
    // remember ile recomposition'larda değer korunur
    var count by remember { mutableStateOf(0) }
    
    Button(onClick = { count++ }) {
        Text("Sayı: $count")  // Doğru çalışır!
    }
}

/*
remember Ne Zaman Temizlenir?
- Composable composition'dan çıkınca
- Parent composable recompose olup bu composable'ı çağırmazsa
- Activity/Fragment destroy olunca

rememberSaveable: 
- Configuration change'de (rotasyon) hayatta kalır
- Process death'de bile hayatta kalır
- Bundle'a serialize edilebilir tipler için
*/

@Composable
fun PersistentCounter() {
    // Ekran döndürülse bile değer korunur
    var count by rememberSaveable { mutableStateOf(0) }
    
    Button(onClick = { count++ }) {
        Text("Sayı: $count")
    }
}

/*
====================================
STATE HOİSTİNG - STATE'İ YUKARI TAŞIMAK
====================================

State Hoisting = State'i üst Composable'a taşımak
Bu Compose'un en önemli pattern'lerinden biri!

Neden?
1. Composable'ı reusable yapar
2. Test edilebilir yapar
3. Single source of truth sağlar
4. Unidirectional data flow sağlar
*/

// KÖTÜ - State içeride, reusable değil
@Composable
fun BadTextField() {
    var text by remember { mutableStateOf("") }
    
    TextField(
        value = text,
        onValueChange = { text = it }
    )
}

// İYİ - State dışarıdan geliyor, reusable
@Composable
fun GoodTextField(
    value: String,
    onValueChange: (String) -> Unit
) {
    TextField(
        value = value,
        onValueChange = onValueChange
    )
}

// Kullanım - State yukarıda kontrol ediliyor
@Composable
fun FormScreen() {
    var name by remember { mutableStateOf("") }
    var email by remember { mutableStateOf("") }
    
    Column {
        GoodTextField(
            value = name,
            onValueChange = { name = it }
        )
        GoodTextField(
            value = email,
            onValueChange = { email = it }
        )
        
        // State'e yukarıdan erişebiliyoruz
        Button(
            onClick = { /* name ve email'i kullan */ },
            enabled = name.isNotEmpty() && email.contains("@")
        ) {
            Text("Kaydet")
        }
    }
}

/*
====================================
COMPOSE PERFORMANS İPUÇLARI
====================================

1. Stable Parameters Kullanın
Compose parametrelerin değişip değişmediğini kontrol eder.
Stable tipler: primitive'ler, String, immutable data class
*/

// Data class'ı stable yapma
@Stable  // veya @Immutable
data class User(
    val id: Int,
    val name: String
)

/*
2. Key Kullanın - Liste elemanlarında
LazyColumn {
    items(users, key = { it.id }) { user ->
        UserCard(user)
    }
}

3. DerivedStateOf - Hesaplanmış değerler için
*/
@Composable
fun SearchScreen() {
    var searchQuery by remember { mutableStateOf("") }
    val userList by remember { mutableStateOf(users) }
    
    // Kötü - her recomposition'da filter çalışır
    // val filteredList = userList.filter { it.name.contains(searchQuery) }
    
    // İyi - sadece searchQuery veya userList değişince hesaplanır
    val filteredList by remember(searchQuery, userList) {
        derivedStateOf {
            userList.filter { it.name.contains(searchQuery) }
        }
    }
}

/*
====================================
COMPOSE vs XML KARŞILAŞTIRMA
====================================

XML/View System:
- 2 dosya (XML + Kotlin)
- findViewById
- View binding/Data binding  
- Manuel state yönetimi
- Boilerplate kod fazla
- Preview sınırlı
- Animasyon zor
- Custom view karmaşık

Compose:
- Tek dosya (Kotlin)
- Direct access
- State-driven UI
- Otomatik güncelleme
- Minimum kod
- Güçlü preview
- Animasyon kolay
- Composition kolay

Örnek: Bir butonun visible/invisible yapılması

XML:
button.visibility = if (condition) View.VISIBLE else View.GONE

Compose:
if (condition) {
    Button(...) { }
}

Compose'da yok olan UI elemanı gerçekten yok!
Memory'den silinir, View.GONE gibi sadece gizlenmez.
*/

/*
====================================
GERÇEK COMPOSE ÖRNEĞİ
====================================
*/

@Composable
fun LoginScreen(
    onLoginClick: (String, String) -> Unit  // State hoisting
) {
    // State tanımlamaları
    var email by remember { mutableStateOf("") }
    var password by remember { mutableStateOf("") }
    var passwordVisible by remember { mutableStateOf(false) }
    var isLoading by remember { mutableStateOf(false) }
    
    // Derived state - hesaplanmış değer
    val isFormValid by remember {
        derivedStateOf {
            email.contains("@") && password.length >= 6
        }
    }
    
    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(16.dp),
        horizontalAlignment = Alignment.CenterHorizontally
    ) {
        // Email input
        OutlinedTextField(
            value = email,
            onValueChange = { email = it },  // State güncelleme
            label = { Text("Email") },
            singleLine = true,
            modifier = Modifier.fillMaxWidth()
        )
        
        Spacer(modifier = Modifier.height(8.dp))
        
        // Password input
        OutlinedTextField(
            value = password,
            onValueChange = { password = it },
            label = { Text("Şifre") },
            visualTransformation = if (passwordVisible) 
                VisualTransformation.None 
            else 
                PasswordVisualTransformation(),
            trailingIcon = {
                IconButton(onClick = { passwordVisible = !passwordVisible }) {
                    Icon(
                        imageVector = if (passwordVisible) 
                            Icons.Default.Visibility 
                        else 
                            Icons.Default.VisibilityOff,
                        contentDescription = null
                    )
                }
            },
            singleLine = true,
            modifier = Modifier.fillMaxWidth()
        )
        
        Spacer(modifier = Modifier.height(16.dp))
        
        // Login button
        Button(
            onClick = { 
                isLoading = true
                onLoginClick(email, password) 
            },
            enabled = isFormValid && !isLoading,  // Derived state kullanımı
            modifier = Modifier.fillMaxWidth()
        ) {
            if (isLoading) {
                CircularProgressIndicator(
                    modifier = Modifier.size(16.dp),
                    color = MaterialTheme.colors.onPrimary
                )
            } else {
                Text("Giriş Yap")
            }
        }
    }
}
